## Extension Point Pattern

```typescript
// extensions/import/plugins/chatgpt.ts
export const ChatGPTSource: ChatSource = {
  name: 'chatgpt',
  version: '1.0.0',
  parse(content: string): ChatSession[] {
    const data = JSON.parse(content);
    return data.conversations.map(conv => ({
      id: conv.id,
      timestamp: new Date(conv.create_time * 1000),
      messages: conv.mapping.map(node => ({
        role: node.message?.author?.role ?? 'system',
        content: node.message?.content?.parts?.join('') ?? '',
        timestamp: new Date(node.message?.create_time * 1000)
      }))
    }));
  }
};
```
