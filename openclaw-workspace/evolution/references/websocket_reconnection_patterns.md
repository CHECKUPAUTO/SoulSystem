## Reconnection Strategy

```typescript
class WebSocketReconnect {
  private ws: WebSocket | null = null;
  private reconnectAttempts = 0;
  private maxReconnectAttempts = 10;
  private baseDelay = 1000; // 1 second
  private maxDelay = 30000; // 30 seconds
  private state: 'connecting' | 'open' | 'closing' | 'closed' = 'closed';

  connect(url: string): void {
    this.state = 'connecting';
    this.ws = new WebSocket(url);
    
    this.ws.onopen = () => {
      this.state = 'open';
      this.reconnectAttempts = 0; // Reset on success
    };
    
    this.ws.onclose = () => {
      this.state = 'closed';
      this.scheduleReconnect(url);
    };
    
    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error);
      // Let onclose handle reconnection
    };
  }

  private scheduleReconnect(url: string): void {
    if (this.reconnectAttempts >= this.maxReconnectAttempts) {
      console.error('Max reconnection attempts reached');
      return;
    }

    const delay = Math.min(
      this.baseDelay * Math.pow(2, this.reconnectAttempts),
      this.maxDelay
    );
    
    // Add jitter to prevent thundering herd
    const jitter = Math.random() * 1000;
    const totalDelay = delay + jitter;
    
    console.log(`Reconnecting in ${totalDelay}ms (attempt ${this.reconnectAttempts + 1})`);
    
    setTimeout(() => {
      this.reconnectAttempts++;
      this.connect(url);
    }, totalDelay);
  }
}
```

## Health Check Pattern

```typescript
// Heartbeat/ping-pong
class WebSocketHealth {
  private lastPong: number = Date.now();
  private heartbeatInterval: NodeJS.Timeout | null = null;
  private pingTimeout = 10000; // 10 seconds

  startHeartbeat(ws: WebSocket): void {
    this.heartbeatInterval = setInterval(() => {
      if (Date.now() - this.lastPong > this.pingTimeout * 2) {
        console.error('Connection stale, forcing reconnect');
        ws.close();
        return;
      }
      ws.send(JSON.stringify({ type: 'ping' }));
    }, this.pingTimeout);
  }

  onPong(): void {
    this.lastPong = Date.now();
  }
}
```

// Gateway client with resilience
class GatewayClient {
  private ws: WebSocketReconnect;
  private messageQueue: any[] = [];

  constructor(private gatewayUrl: string) {
    this.ws = new WebSocketReconnect();
  }

  async connect(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws.connect(this.gatewayUrl);
      // Resolve on open, reject on max retries
    });
  }

  send(message: any): void {
    if (this.ws.state === 'open') {
      this.ws.send(JSON.stringify(message));
    } else {
      this.messageQueue.push(message);
    }
  }
}
